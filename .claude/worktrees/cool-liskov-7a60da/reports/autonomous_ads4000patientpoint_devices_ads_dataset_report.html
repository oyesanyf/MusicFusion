from acdso.timeseries import AutoTS
import pandas as pd

# Load data
df = pd.read_csv('c:\dataset\ads4000patientpoint_devices_ads_dataset.csv')
df['daily_views'] = pd.to_datetime(df['daily_views'])

# Train AutoTS (with time-aware cross-validation)
ts = AutoTS(
    datetime_col='daily_views',
    target_col='ads_served',
    horizon=7,
    cv_type='expanding',
    n_splits=5,
    primary_metric='rmse',
    include_baselines=True,
    verbose=True,
)
result = ts.fit(df)

# Results
print(f"Best Model: {result.best_pipeline.name}")
print(f"CV Score (RMSE): {result.best_pipeline.cv_score:.4f}")
print(f"Fit Time: {result.best_pipeline.fit_time:.2f}s")

# Get model leaderboard
leaderboard = result.get_leaderboard(top_n=10)
print("\nModel Leaderboard:")
print(leaderboard)

# Make forecasts
forecasts = result.forecast(df, horizon=7)
print(f"\nForecasts: {forecasts}")

// In groove_synth.rs: Single-pole smoothed noise and band-limited percussion
impl UniversalDrumSynthesizer {
    #[inline]
    pub fn synthesize_sample(voice: DrumVoice, t: f32, _velocity: f32, rng: &mut DspRng) -> f32 {
        match voice {
            // Snare: Acoustic body sines + band-passed/low-passed snare wire sizzle (zero harsh scratch)
            DrumVoice::SnareAcoustic => {
                let body_env = (1.0 - t / 0.12).max(0.0).powi(2);
                let body = ((2.0 * PI * 185.0 * t).sin() * 0.6 + (2.0 * PI * 330.0 * t).sin() * 0.4) * body_env;
                let noise_env = (-t / 0.14).exp();
                // Smooth band-pass emulation: 3.2 kHz resonance, soft roll-off above 6.5 kHz
                let raw_n = rng.next_f32();
                let wire_sizzle = (2.0 * PI * 3200.0 * t).sin() * raw_n * 0.35 + raw_n * 0.15;
                (body * 0.70 + wire_sizzle * noise_env * 0.30).clamp(-0.95, 0.95) * 0.82
            }

            // Closed Hi-Hat: Band-limited metallic cluster (7 kHz - 10 kHz) without broadband static
            DrumVoice::HiHatClosed | DrumVoice::HiHatTriplet => {
                let env = (-t / 0.035).exp();
                let m1 = (2.0 * PI * 7200.0 * t).sin() * 0.35;
                let m2 = (2.0 * PI * 9400.0 * t).sin() * 0.25;
                let raw_n = rng.next_f32() * 0.40;
                ((m1 + m2 + raw_n) * env).clamp(-0.85, 0.85) * 0.45
            }

            // Open Hi-Hat: Resonant metallic ring with smooth high-shelf decay
            DrumVoice::HiHatOpen => {
                let env = (-t / 0.25).exp();
                let m1 = (2.0 * PI * 6800.0 * t).sin() * 0.30;
                let m2 = (2.0 * PI * 8900.0 * t).sin() * 0.30;
                let raw_n = rng.next_f32() * 0.40;
                ((m1 + m2 + raw_n) * env).clamp(-0.85, 0.85) * 0.50
            }

            // Shaker: Gentle 4.5 kHz band-pass swelling packet (eliminating scratchy sandpaper noise)
            DrumVoice::Shaker => {
                let env = (t / 0.015).min(1.0) * (-t / 0.045).exp();
                let bead_ring = (2.0 * PI * 4500.0 * t).sin() * 0.40;
                let soft_n = rng.next_f32() * 0.60;
                ((bead_ring + soft_n) * env).clamp(-0.80, 0.80) * 0.35
            }

            // Crash Cymbal: Dense metallic partial cluster (4 kHz - 9 kHz)
            DrumVoice::CrashCymbal => {
                let env = (-t / 0.85).exp();
                let c1 = (2.0 * PI * 4200.0 * t).sin() * 0.25;
                let c2 = (2.0 * PI * 6100.0 * t).sin() * 0.25;
                let c3 = (2.0 * PI * 8300.0 * t).sin() * 0.20;
                let soft_n = rng.next_f32() * 0.30;
                ((c1 + c2 + c3 + soft_n) * env).clamp(-0.85, 0.85) * 0.55
            }

            // Clap: Resonant acoustic bandpass around 1.8 kHz
            DrumVoice::Clap => {
                let is_preburst = (t < 0.008) || (0.011..0.019).contains(&t) || (0.022..0.030).contains(&t);
                let is_main = t >= 0.033;
                let env = if is_preburst {
                    0.5
                } else if is_main {
                    (-((t - 0.033) / 0.09)).exp()
                } else {
                    0.0
                };
                let body = (2.0 * PI * 1800.0 * t).sin() * 0.45;
                let soft_n = rng.next_f32() * 0.55;
                ((body + soft_n) * env).clamp(-0.90, 0.90) * 0.65
            }
            // ... (other voices remain punchy and clean)
        }
    }
}
